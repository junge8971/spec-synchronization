Шаг 1 - запрос передачи фильтров/создания выборки по условиям отбора
| POST: /api/dialogs/conditions
| Description | Создать выборку диалогов по условиям отбора. Нужен, когда условий много и они не влезают в URL. Возвращает ID выборки для дальнейшего запроса.
|
parameters
|
-
|
Request Body
|
{
"sudir_id": "id", //обязательное - идентификатор пользователя sudir , если аноним - null
"access_token": "токен", //обязательно - токен пользователя sudir, если аноним - null
//Условия отбора
"match": "and", //and — все условия, or — любое условие
"conditions": [
{
"field": "phrase", //имя атрибута
"operator": "contains", //встречается, не встречается
"value": null, //значение для сравнения
"searchMode": "words", //words — слова в любом порядке, exact — точно как написано
"scope": "all" //all — весь диалог, user — только пользователь, assistant — только ассистент, skill / operator - если надо выбирать
"skillId": "id" //если scope = skill
"operatorId": "id" //если scope = operator
}
]
}

| 2 | firstUserMessage | string | Здравствуйте, мы переехали в новую квартиру на прошлой неделе... | Первое сообщение пользователя (тема диалога)
| 3 | participants | array[object] | [{ "type": "skill", "id": "s-water", "name": "Вода" }, ...] | Участники диалога. type: skill / operator, id: идентификатор, name: название/ФИО

| 8 | endedAt | string (ISO 8601) или null | 2026-08-12T15:23:34+03:00 | Дата завершения диалога, null если активен
| 9 | durationSec | integer | 1284 | Длительность диалога в секундах
| 10 | channel | string | Портал mos.ru | Канал связи (локализованное название)
| 11 | operatorRating | integer (1–5) или null | 3 | Оценка оператора, null если нет
| 12 | hasIncidentMark | boolean | false | Есть ли отметка об инциденте

| # | Поле | Тип | Пример | Описание
| 1 | total | integer | 52 | Всего диалогов по текущим фильтрам

| # | Поле | Тип | Пример | Описание

| 2 | shown | integer | 25 | Отображено на текущей странице
| 3 | offset | integer | 0 | Текущее смещение
| 4 | limit | integer | 25 | Размер страницы
| 5 | hasMore | boolean | true | Есть ли ещё записи для подгрузки
