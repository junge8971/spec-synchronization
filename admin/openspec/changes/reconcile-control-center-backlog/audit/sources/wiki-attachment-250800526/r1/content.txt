function plural(n, forms){
  const a = Math.abs(n) % 100, b = a % 10;
  if (a > 10 && a < 20) return forms[2];
  if (b > 1 && b < 5) return forms[1];
  if (b === 1) return forms[0];
  return forms[2];
}
function esc(s){
  return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
  }[c]));
}

function findFieldDef(condField){
  for (const key in AF_FIELDS){
    const f = AF_FIELDS[key].find(x => x.id === condField);
    if (f) return f;
  }
  return null;
}
function matchesCondition(record, cond){
  const fieldDef = findFieldDef(cond.field);
  if (!fieldDef) return true;
  const value = fieldDef.get(record);
  const op = cond.op, v = cond.value;
  if (fieldDef.type === 'enum'){
    const arr = Array.isArray(value) ? value : [value];
    const sel = Array.isArray(v) ? v : (v ? [v] : []);
    if (op === 'in') return sel.length === 0 || sel.some(s => arr.includes(s));
    if (op === 'notIn') return sel.length === 0 || !sel.some(s => arr.includes(s));
    return true;
  }
  const s = String(value == null ? '' : value).toLowerCase();
  if (op === 'empty')    return s.trim() === '';
  if (op === 'notEmpty') return s.trim() !== '';
  if (v == null || String(v).trim() === '') return true;
  const needle = String(v).toLowerCase();
  if (op === 'contains')    return s.includes(needle);
  if (op === 'notContains') return !s.includes(needle);
  if (op === 'equals')      return s === needle;
  if (op === 'notEquals')   return s !== needle;
  if (op === 'startsWith')  return s.startsWith(needle);
  if (op === 'endsWith')    return s.endsWith(needle);
  return true;
}
function matchesAdvanced(record, listId){
  const state = advFilters[listId];
  if (!state) return true;
  const conds = state.appliedConditions;
  if (!conds || conds.length === 0) return true;
  const results = conds.map(c => matchesCondition(record, c));
  return state.appliedJoin === 'or' ? results.some(Boolean) : results.every(Boolean);
}

function renderFilterConditions(listId){
  const state = advFilters[listId];
  const fields = AF_FIELDS[listId];
  const wrap = document.getElementById('filter-conditions-' + listId);
  if (!wrap) return;
  wrap.innerHTML = '';
  if (state.conditions.length === 0){
    wrap.innerHTML = '<div class="filter-empty">Условий нет. Нажмите «Добавить условие».</div>';
    return;
  }
  state.conditions.forEach(c => {
    const fieldDef = fields.find(f => f.id === c.field) || fields[0];
    const ops = AF_OPS[fieldDef.type];
    const opDef = ops.find(o => o.id === c.op) || ops[0];
    const row = document.createElement('div');
    row.className = 'filter-row';

    const fieldSel = document.createElement('select');
    fieldSel.className = 'input';
    fields.forEach(f => {
      const opt = document.createElement('option');
      opt.value = f.id; opt.textContent = f.label;
      if (f.id === fieldDef.id) opt.selected = true;
      fieldSel.appendChild(opt);
    });
    fieldSel.onchange = () => onFilterFieldChange(listId, c.id, fieldSel.value);

    const opSel = document.createElement('select');
    opSel.className = 'input';
    ops.forEach(o => {
      const opt = document.createElement('option');
      opt.value = o.id; opt.textContent = o.label;
      if (o.id === opDef.id) opt.selected = true;
      opSel.appendChild(opt);
    });
    opSel.onchange = () => onFilterOpChange(listId, c.id, opSel.value);

    const valWrap = document.createElement('div');
    valWrap.className = 'filter-value-wrap';
    if (!opDef.noValue){
      if (fieldDef.type === 'enum'){
        const chips = document.createElement('div');
        chips.className = 'filter-enum-chips';
        const selected = Array.isArray(c.value) ? c.value : [];
        fieldDef.options.forEach((code, i) => {
          const lbl = document.createElement('label');
          const on = selected.includes(code);
          lbl.className = 'filter-enum-chip' + (on ? ' on' : '');
          lbl.innerHTML = `<input type="checkbox" ${on ? 'checked' : ''} /><span>${esc(fieldDef.labels[i])}</span>`;
          lbl.onclick = (e) => {
            e.preventDefault();
            const arr = Array.isArray(c.value) ? c.value.slice() : [];
            const ix = arr.indexOf(code);
            if (ix >= 0) arr.splice(ix,1); else arr.push(code);
            c.value = arr;
            renderFilterConditions(listId);
          };
          chips.appendChild(lbl);
        });
        valWrap.appendChild(chips);
      } else {
        const inp = document.createElement('input');
        inp.className = 'input filter-value';
        inp.placeholder = 'значение';
        inp.value = c.value ?? '';
        inp.oninput = () => { c.value = inp.value; };
        valWrap.appendChild(inp);
      }
    } else {
      const note = document.createElement('span');
      note.style.color = 'var(--muted)';
      note.style.fontSize = '11.5px';
      note.textContent = 'значение не требуется';
      valWrap.appendChild(note);
    }
    const del = document.createElement('button');
    del.type = 'button'; del.className = 'filter-del'; del.textContent = '✕';
    del.onclick = () => removeFilterCondition(listId, c.id);

    row.appendChild(fieldSel); row.appendChild(opSel); row.appendChild(valWrap); row.appendChild(del);
    wrap.appendChild(row);
  });
}
function addFilterCondition(listId){
  const state = advFilters[listId];
  const fields = AF_FIELDS[listId];
  const firstField = fields[0];
  const firstOp = AF_OPS[firstField.type][0];
  state.conditions.push({
    id: newAfId(), field: firstField.id, op: firstOp.id,
    value: firstField.type === 'enum' ? [] : ''
  });
  renderFilterConditions(listId);
}
function removeFilterCondition(listId, condId){
  const state = advFilters[listId];
  state.conditions = state.conditions.filter(c => c.id !== condId);
  renderFilterConditions(listId);
}
function onFilterFieldChange(listId, condId, fieldId){
  const state = advFilters[listId];
  const c = state.conditions.find(x => x.id === condId);
  if (!c) return;
  const fieldDef = AF_FIELDS[listId].find(f => f.id === fieldId);
  const firstOp = AF_OPS[fieldDef.type][0];
  c.field = fieldId; c.op = firstOp.id;
  c.value = fieldDef.type === 'enum' ? [] : '';
  renderFilterConditions(listId);
}
function onFilterOpChange(listId, condId, opId){
  const state = advFilters[listId];
  const c = state.conditions.find(x => x.id === condId);
  if (!c) return;
  c.op = opId;
  renderFilterConditions(listId);
}

function describeConditions(listId, conds, join){
  const fields = AF_FIELDS[listId];
  return conds.map(c => {
    const fieldDef = fields.find(f => f.id === c.field);
    if (!fieldDef) return '';
    const opDef = AF_OPS[fieldDef.type].find(o => o.id === c.op);
    let val = '';
    if (Array.isArray(c.value)){
      val = c.value.map(code => {
        const i = fieldDef.options.indexOf(code);
        return fieldDef.labels ? fieldDef.labels[i] : code;
      }).join(', ');
    } else val = c.value ?? '';
    return `${fieldDef.label} ${opDef ? opDef.label : c.op}${val ? ' ' + val : ''}`;
  }).filter(Boolean).join(join === 'and' ? ' и ' : ' или ');
}

}

    <div style="padding:8px 10px 4px;font-size:11px;color:var(--muted);border-top:1px solid var(--border);margin-top:4px;letter-spacing:.05em;text-transform:uppercase;font-weight:600">
      Сохранённые фильтры
    </div>

      ? `<div style="padding:6px 10px 10px;font-size:12px;color:var(--muted);font-style:italic;line-height:1.45">

        </div>`).join('')}
  `;
}
function saveCurrentFilter(listId){
  const name = window.prompt('Название фильтра');
  if (!name || !name.trim()) return;
  const trimmed = name.trim();
  const filterData = {
    id: newSfId(), name: trimmed,
    search: (document.getElementById('api-q')?.value || ''),
    ms: {}, af: null
  };
  Object.keys(msState).forEach(k => {
    if (k.startsWith(listId) && msState[k].length > 0) filterData.ms[k] = msState[k].slice();
  });
  if (advFilters[listId]) filterData.af = {
    conditions: JSON.parse(JSON.stringify(advFilters[listId].appliedConditions)),
    join: advFilters[listId].appliedJoin
  };
  const arr = savedFilters[listId];
  const idx = arr.findIndex(f => f.name === trimmed);
  if (idx >= 0) arr[idx] = filterData; else arr.push(filterData);
  toast(`Фильтр «${trimmed}» сохранён`);
  document.querySelectorAll('.menu-content.open').forEach(m => m.classList.remove('open'));
}
function applySavedFilter(listId, filterId){
  const f = (savedFilters[listId] || []).find(x => x.id === filterId);
  if (!f) return;
  const s = document.getElementById('api-q');
  if (s) s.value = f.search || '';
  Object.keys(msState).forEach(k => {
    if (k.startsWith(listId)){
      msState[k] = (f.ms[k] || []).slice();
      document.querySelectorAll('#' + k + '-menu input[type=checkbox]').forEach(cb => {
        cb.checked = msState[k].includes(cb.value);
      });
      updateMsLabel(k);
    }
  });
  if (f.af){
    advFilters[listId].appliedConditions = JSON.parse(JSON.stringify(f.af.conditions));
    advFilters[listId].appliedJoin = f.af.join;
    updateAfBadge(listId); renderAppliedBar(listId);
  }
  renderApiList();
  updateResetVisibility(listId);
  toast(`Фильтр «${f.name}» применён`);
  document.querySelectorAll('.menu-content.open').forEach(m => m.classList.remove('open'));
}
function deleteSavedFilter(listId, filterId){
  savedFilters[listId] = (savedFilters[listId] || []).filter(f => f.id !== filterId);
  renderMenuContent(listId);
  toast('Фильтр удалён');
}
function copyFilterLink(listId){
  const search = document.getElementById('api-q')?.value.trim() || '';
  const params = new URLSearchParams();
  if (search) params.set('q', search);
  if (msState['api-method']?.length) params.set('method', msState['api-method'].join(','));
  if (msState['api-source']?.length) params.set('source', msState['api-source'].join(','));
  if (msState['api-status']?.length) params.set('status', msState['api-status'].join(','));
  const url = window.location.origin + window.location.pathname + '?' + params.toString();
  if (navigator.clipboard) navigator.clipboard.writeText(url);
  toast('Ссылка на отбор скопирована');
  document.querySelectorAll('.menu-content.open').forEach(m => m.classList.remove('open'));
}

let apiSort = { field:null, dir:null };
function setupSorting(){
  document.querySelectorAll('#screen-api-list thead th.sortable').forEach(th => {
    th.onclick = () => {
      const f = th.dataset.sort;
      if (apiSort.field === f){
        if (apiSort.dir === 'asc') apiSort.dir = 'desc';
        else { apiSort.field = null; apiSort.dir = null; }
      } else { apiSort.field = f; apiSort.dir = 'asc'; }
      paintSort('#screen-api-list thead', apiSort);
      renderApiList();
    };
  });
}
function paintSort(sel, sort){
  document.querySelectorAll(sel + ' th.sortable').forEach(th => {
    if (th.dataset.sort === sort.field) th.dataset.dir = sort.dir || '';
    else th.removeAttribute('data-dir');
  });
}
function applySort(rows, sort, accessors){
  if (!sort.field || !sort.dir) return rows;
  const acc = accessors[sort.field];
  if (!acc) return rows;
  const dir = sort.dir === 'asc' ? 1 : -1;
  return rows.slice().sort((a,b) => {
    const va = acc(a), vb = acc(b);
    if (va == null && vb == null) return 0;
    if (va == null) return 1;
    if (vb == null) return -1;
    if (typeof va === 'number' && typeof vb === 'number') return (va - vb) * dir;
    return String(va).localeCompare(String(vb), 'ru') * dir;
  });
}

  const s = String(m || '').toLowerCase();

function renderApiList(){
  const q  = document.getElementById('api-q').value.trim().toLowerCase();
  const mSel = msState['api-method'];
  const sSel = msState['api-source'];
  const stSel = msState['api-status'];

  const filtered = API_DATA.filter(r => {
    if (apiGroupTab !== 'all' && r.group_code !== apiGroupTab) return false;
    if (mSel.length && !mSel.includes(r.method)) return false;
    if (sSel.length && !sSel.includes(r.source_code || '')) return false;
    if (stSel.length && !stSel.includes(r.status)) return false;
    if (q.length >= 2){
      const hay = (r.name + ' ' + r.code + ' ' + r.path + ' ' + r.description).toLowerCase();
      if (!hay.includes(q)) return false;
    }
    if (!matchesAdvanced(r, 'api')) return false;
    return true;
  });
  const rows = applySort(filtered, apiSort, {
    method: r => r.method,
    path: r => r.path,
    group_code: r => labelOf(DICT_API_GROUPS, r.group_code) + (r.source_code ? ' · ' + labelOf(DICT_API_SOURCES, r.source_code) : ''),
    version: r => r.version,
    status: r => labelOf(DICT_API_STATUSES, r.status),
    name: r => r.name
  });

  const tbody = document.getElementById('api-rows');
  const empty = document.getElementById('api-empty');
  tbody.innerHTML = '';
  empty.style.display = rows.length === 0 ? 'block' : 'none';

  rows.forEach(r => {
    const tr = document.createElement('tr');
    tr.onclick = () => openApiCard(r.id);
    const stTag = r.status === 'active' ? 'ok'
                : r.status === 'deprecated' ? 'warn'
                : r.status === 'archived' ? 'mute' : 'ink';
    const groupLine = labelOf(DICT_API_GROUPS, r.group_code) + (r.source_code ? ' · ' + labelOf(DICT_API_SOURCES, r.source_code) : '');
    tr.innerHTML = `
      <td>${methodBadge(r.method)}</td>
      <td>
        <div class="name-cell">
          <span class="n">${esc(r.name)}</span>
          <span class="c endpoint-path">${esc(r.path)}</span>
        </div>
      </td>
      <td>${esc(groupLine)}</td>
      <td><span class="num mono" style="font-size:12px">${esc(r.version || '—')}</span></td>
      <td><span class="tag ${stTag}">${esc(labelOf(DICT_API_STATUSES, r.status))}</span></td>
      <td class="right">›</td>
    `;
    tbody.appendChild(tr);
  });

  document.getElementById('api-count').textContent =
    rows.length + ' ' + plural(rows.length, ['эндпоинт','эндпоинта','эндпоинтов']);

  updateApiTabCounts();
}

  apiParams = apiParams.filter(p => p._id !== id);
  renderApiParams();
  toast('Параметр удалён');
}
function updateApiParam(id, key, value){
  const p = apiParams.find(x => x._id === id);
  if (!p) return;
  p[key] = value;
}

function attachApiParamDrag(row, id){
  row.addEventListener('dragstart', e => {
    apiParamDragId = id;
    row.style.opacity = '.4';
    e.dataTransfer.effectAllowed = 'move';
    try { e.dataTransfer.setData('text/plain', id); } catch(_){}
  });
  row.addEventListener('dragend', () => {
    row.style.opacity = '';
    document.querySelectorAll('.api-edit-row').forEach(el => el.style.borderColor = '');
    apiParamDragId = null;
  });
  row.addEventListener('dragover', e => {
    e.preventDefault();
    if (apiParamDragId && apiParamDragId !== id) row.style.borderColor = 'var(--accent)';
  });
  row.addEventListener('dragleave', () => row.style.borderColor = '');
  row.addEventListener('drop', e => {
    e.preventDefault();
    row.style.borderColor = '';
    if (!apiParamDragId || apiParamDragId === id) return;
    const from = apiParams.findIndex(x => x._id === apiParamDragId);
    const to = apiParams.findIndex(x => x._id === id);
    if (from < 0 || to < 0) return;
    const [moved] = apiParams.splice(from, 1);
    apiParams.splice(to, 0, moved);
    apiParamDragId = null;
    renderApiParams();
  });
}

function renderApiResponses(){
  const wrap = document.getElementById('api-responses-list');
  if (!wrap) return;
  wrap.innerHTML = '';
  if (apiResponses.length === 0){
    wrap.innerHTML = `<div class="empty" style="padding:18px">Ответов пока нет. Добавьте первый ниже.</div>`;
    renderApiResponsePreview();
    return;
  }
  if (!apiResponses.some(r => r._id === apiSelectedResponseId)){
    apiSelectedResponseId = apiResponses[0]._id;
  }
  apiResponses.forEach(r => {
    const row = document.createElement('div');
    row.className = 'api-resp-row';
    row.style.cursor = 'pointer';
    row.dataset.id = r._id;
    if (r._id === apiSelectedResponseId){
      row.style.borderColor = 'var(--accent)';
      row.style.background = '#fffafa';
    }
    row.onclick = (e) => {
      if (e.target.closest('input, button')) return;
      apiSelectedResponseId = r._id;
      renderApiResponses();
    };
    row.innerHTML = `
      <input class="input" value="${esc(r.http_code || '')}"
             placeholder="200" oninput="updateApiResponse('${r._id}','http_code',this.value)" />
      <input class="input" value="${esc(r.description || '')}"
             placeholder="описание" oninput="updateApiResponse('${r._id}','description',this.value)" />
      <input class="input url-input" value="${esc(r.example || '')}"
             placeholder='{ "ok": true }' oninput="updateApiResponse('${r._id}','example',this.value)" />
      <button class="field-del" title="Удалить" onclick="removeApiResponse('${r._id}')">✕</button>
    `;
    wrap.appendChild(row);
  });
  renderApiResponsePreview();
}

function addApiResponse(){
  const r = { _id:'r' + Date.now() + '_' + (apiResponses.length+1),
              http_code:'200', description:'Описание', example:'{ "ok": true }' };
  apiResponses.push(r);
  apiSelectedResponseId = r._id;
  renderApiResponses();
  toast('Ответ добавлен');
}
function removeApiResponse(id){
  apiResponses = apiResponses.filter(p => p._id !== id);
  if (apiSelectedResponseId === id) apiSelectedResponseId = apiResponses.length ? apiResponses[0]._id : null;
  renderApiResponses();
  toast('Ответ удалён');
}
function updateApiResponse(id, key, value){
  const r = apiResponses.find(x => x._id === id);
  if (!r) return;
  r[key] = value;
  if (apiSelectedResponseId === id) renderApiResponsePreview();
}

function renderApiResponsePreview(){
  const el = document.getElementById('api-response-preview');
  if (!el) return;
  const r = apiResponses.find(x => x._id === apiSelectedResponseId) || apiResponses[0];
  if (!r){
    el.textContent = 'Здесь появится пример тела ответа.';
    el.style.color = 'rgba(232,235,240,.5)';
    return;
  }
  el.style.color = '';
  el.textContent = (r.http_code ? 'HTTP ' + r.http_code + (r.description ? ' — ' + r.description : '') + '\n\n' : '') + (r.example || '');
}

function clearApiErrors(){
  ['api-name','api-code'].forEach(id => {
    const wrap = document.getElementById(id + '-field');
    if (wrap) wrap.classList.remove('has-error');
  });
}
function checkApiUnique(key){
  const value = (draft[key] || '').trim();
  const wrap  = document.getElementById('api-' + key + '-field');
  if (!wrap) return true;
  let ok = true;
  if (!value) ok = false;
  else {
    const dup = API_DATA.some(d => d[key] === value && d.id !== currentApiId);
    if (dup) ok = false;
  }
  wrap.classList.toggle('has-error', !ok);
  return ok;
}
function saveApiCard(){
  const okName = checkApiUnique('name');
  const okCode = checkApiUnique('code');
  const pathVal = (draft.path || '').trim();
  if (!okName || !okCode || !pathVal){
    toast('Проверьте обязательные поля');
    return;
  }
  draft.updated_at = new Date().toISOString().slice(0,16).replace('T',' ');
  draft.params = apiParams.map(p => ({
    location: p.location, name: p.name, data_type: p.data_type,
    required: !!p.required, description: p.description
  }));
  draft.responses = apiResponses.map(r => ({
    http_code: r.http_code, description: r.description, example: r.example
  }));

  if (currentApiId){
    const ix = API_DATA.findIndex(d => d.id === currentApiId);
    if (ix >= 0) API_DATA[ix] = JSON.parse(JSON.stringify(draft));
  } else {
    draft.id = 'd' + Date.now();
    API_DATA.push(JSON.parse(JSON.stringify(draft)));
    currentApiId = draft.id;
  }
  renderApiList();
  renderHead();
  toast('Изменения сохранены');
}
function pauseApiCard(){
  if (!draft) return;
  draft.status = draft.status === 'paused' ? 'active' : 'paused';
  fillSelect(document.getElementById('api-status'), DICT_API_STATUSES, draft.status);
  renderHead();
  toast(draft.status === 'paused' ? 'Эндпоинт приостановлен' : 'Эндпоинт возобновлён');
}
function archiveApiCard(){
  if (!draft) return;
  if (!confirm('Перевести эндпоинт в архив?')) return;
  draft.status = 'archived';
  fillSelect(document.getElementById('api-status'), DICT_API_STATUSES, draft.status);
  renderHead();
  toast('Эндпоинт переведён в архив');
}
