function esc(s){
  return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
  }[c]));
}
function toast(msg){
  const el = document.getElementById('toast');
  el.textContent = msg;
  el.classList.add('on');
  clearTimeout(toast._t);
  toast._t = setTimeout(() => el.classList.remove('on'), 2200);
}
function plural(n, forms){
  const a = Math.abs(n) % 100, b = a % 10;
  if (a > 10 && a < 20) return forms[2];
  if (b > 1 && b < 5) return forms[1];
  if (b === 1) return forms[0];
  return forms[2];
}

function toggleMs(id){
  const menu = document.getElementById(id + '-menu');
  const isOpen = menu.classList.contains('open');
  document.querySelectorAll('.ms-menu.open').forEach(m => m.classList.remove('open'));
  document.querySelectorAll('.ms-btn.active').forEach(b => b.classList.remove('active'));
  if (!isOpen){
    menu.classList.add('open');
    document.getElementById(id + '-btn').classList.add('active');
  }
}
function clearMs(id){
  msState[id] = [];

function renderList(){
  const q = document.getElementById('q').value.trim().toLowerCase();
  const kindSel = msState['ref-kind'];
  const statusSel = msState['ref-status'];

  const filtered = DICTIONARIES.filter(d => {
    if (kindSel.length && !kindSel.includes(d.type)) return false;
    if (statusSel.length && !statusSel.includes(d.status)) return false;
    if (q.length >= 2){
      const hay = (d.name + ' ' + d.code + ' ' + d.description).toLowerCase();
      if (!hay.includes(q)) return false;
    }
    return true;
  });

  const tbody = document.getElementById('rows');
  const empty = document.getElementById('empty');
  tbody.innerHTML = '';
  empty.style.display = filtered.length === 0 ? 'block' : 'none';

  filtered.forEach(d => {
    const tr = document.createElement('tr');
    tr.onclick = () => openCard(d.id);
    const statusTag = d.status === 'Активен' ? 'ok'
                    : d.status === 'Приостановлен' ? 'warn'
                    : d.status === 'Черновик' ? 'ink' : 'mute';
    const typeTag = d.type === 'Системный' ? 'sys' : 'custom';
    const usedIn = d.usage.slice(0,2).map(u => u.title).join(' · ') +
                   (d.usage.length > 2 ? ` и ещё ${d.usage.length - 2}` : '');
    const kindMark = d.kind === 'rules-attachments'
      ? `<span class="sys-row-badge custom" title="Правила вложений">📎 правила</span>`
      : '';

    tr.innerHTML = `
      <td class="name-cell">
        <span class="n">${esc(d.name)} ${kindMark}</span>
        <span class="c">${esc(d.code)}</span>
      </td>
      <td><span class="tag ${typeTag}">${esc(d.type)}</span></td>
      <td class="right num">${d.values.length}</td>
      <td style="color:var(--muted);font-size:12px">${esc(usedIn)}</td>
      <td><span class="tag ${statusTag}">${esc(d.status)}</span></td>
      <td class="right">›</td>
    `;
    tbody.appendChild(tr);
  });

  document.getElementById('count').textContent =
    filtered.length + ' ' + plural(filtered.length, ['справочник','справочника','справочников']);
}

  ev?.stopPropagation();

/* Drag & drop для строк значений */
function attachRowDrag(tr, id){
  tr.addEventListener('dragstart', e => {
    dragId = id;
    tr.classList.add('dragging');
    e.dataTransfer.effectAllowed = 'move';
    try { e.dataTransfer.setData('text/plain', id); } catch(_){}
  });
  tr.addEventListener('dragend', () => {
    tr.classList.remove('dragging');
    document.querySelectorAll('.values-table tr.drop-target').forEach(el => el.classList.remove('drop-target'));
    dragId = null;
  });
  tr.addEventListener('dragover', e => {
    e.preventDefault();
    e.dataTransfer.dropEffect = 'move';
    if (dragId && dragId !== id) tr.classList.add('drop-target');
  });
  tr.addEventListener('dragleave', () => tr.classList.remove('drop-target'));
  tr.addEventListener('drop', e => {
    e.preventDefault();
    tr.classList.remove('drop-target');
    if (!dragId || dragId === id) return;
    const d = getCurrent();
    if (!d) return;
    const from = d.values.findIndex(x => x.id === dragId);
    const to = d.values.findIndex(x => x.id === id);
    if (from < 0 || to < 0) return;
    const [moved] = d.values.splice(from, 1);
    d.values.splice(to, 0, moved);
    dragId = null;
    markDirty();
    renderValues();
  });
}

  // Обновляем шапку
  document.getElementById('title').textContent = d.name;
  document.getElementById('crumb-name').textContent = d.name;
  document.getElementById('head-code').textContent = d.code;
  document.getElementById('head-updated').textContent = d.updatedAt;
  document.getElementById('subtitle').textContent = d.description;
  document.getElementById('btn-add-value').disabled = !d.allowAdd;

  renderList();
  renderValues();
  renderValuesBanner();
  toast('Изменения сохранены');
}

function togglePauseDictionary(){
  const d = getCurrent();
  if (!d) return;
  d.status = d.status === 'Приостановлен' ? 'Активен' : 'Приостановлен';
  d.updatedAt = nowStr();
  document.getElementById('head-status-text').textContent = d.status;
  const dot = document.getElementById('head-status-dot');
  if (d.status === 'Активен') dot.classList.remove('off'); else dot.classList.add('off');
  document.getElementById('btn-pause').textContent = d.status === 'Приостановлен' ? 'Возобновить' : 'Приостановить';
  document.getElementById('head-updated').textContent = d.updatedAt;
  renderList();
  toast(d.status === 'Активен' ? 'Справочник возобновлён' : 'Справочник приостановлен');
}

function deleteDictionary(){
  const d = getCurrent();
  if (!d) return;
  if (d.type === 'Системный'){
    toast('Системный справочник нельзя удалить');
    return;
  }
  if (!confirm(`Удалить справочник «${d.name}»? Значения перестанут быть доступными в фильтрах и формах.`)) return;
  const idx = DICTIONARIES.findIndex(x => x.id === d.id);
  if (idx >= 0) DICTIONARIES.splice(idx, 1);
  toast('Справочник удалён');
  showSection('ref','list');
  renderList();
}

function createNewDictionary(){
  const name = prompt('Название нового справочника');
  if (!name || !name.trim()) return;
  const code = prompt('Служебный код (латиницей, например: my_dictionary)');
  if (!code || !code.trim()) return;

  const id = 'd' + Date.now();
  DICTIONARIES.unshift({
    id,
    name: name.trim(),
    code: code.trim(),
    description: '',
    kind:'list',
    type:'Пользовательский',
    owner:'Команда ЖКХ',
    status:'Черновик',
    updatedAt: nowStr(),
    publicApi:false,
    allowAdd:true,
    allowDelete:true,
    allowRename:true,
    required:false,
    cache:0,
    usage:[],
    values:[]
  });
  renderList();
  openCard(id);
  toast('Справочник создан. Добавьте значения во вкладке «Значения».');
}

function nowStr(){
  const d = new Date();
  const pad = n => String(n).padStart(2,'0');
  return d.getFullYear() + '-' + pad(d.getMonth()+1) + '-' + pad(d.getDate()) +
    ' ' + pad(d.getHours()) + ':' + pad(d.getMinutes());
}

function exportValuesCsv(){
  const d = getCurrent();
  if (!d) return;
  const rows = d.kind === 'rules-attachments'
    ? [['Код','Категория','Расширения','MIME','Макс. размер, МБ','Макс. кол-во','Зоны','Включено','Примечание'],
       ...d.values.map(v => [v.code, v.label, v.extensions, v.mime, v.maxSizeMb, v.maxCount, v.zones, v.enabled ? 'да' : 'нет', v.note || ''])]
    : [['Код','Отображается как','Описание','Цвет','По умолчанию','Включено'],
       ...d.values.map(v => [v.code, v.label, v.description || '', v.color || '', v.isDefault ? 'да' : 'нет', v.enabled ? 'да' : 'нет'])];

  const csv = rows.map(r => r.map(c => {
    const s = String(c ?? '');
    return /[",\n;]/.test(s) ? `"${s.replace(/"/g,'""')}"` : s;
  }).join(';')).join('\n');

  const blob = new Blob(['\uFEFF' + csv], { type:'text/csv;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `${d.code}.csv`;
  a.click();
  URL.revokeObjectURL(url);
  toast('CSV выгружен');
}

function showSection(section, view){

  if (el) el.classList.add('active');
  const crumbNow = document.getElementById('crumb-now');
  if (view === 'card'){
    crumbNow.textContent = document.getElementById('title').textContent;
  } else {
    crumbNow.textContent = 'Справочники';
    // Сброс открытого справочника при возврате
    if (dirty) dirty = false;
  }
  window.scrollTo(0,0);
}
