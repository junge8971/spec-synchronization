function matchesCondition(record, cond){
  let fieldDef = null;
  for (const key in AF_FIELDS){
    fieldDef = AF_FIELDS[key].find(f => f.id === cond.field);
    if (fieldDef) break;
  }
  if (!fieldDef) return true;
  const value = fieldDef.get(record);
  const op = cond.op;
  const v = cond.value;
  const v2 = cond.value2;

  if (fieldDef.type === 'enum'){
    const arr = Array.isArray(value) ? value : [value];
    const sel = Array.isArray(v) ? v : (v ? [v] : []);
    if (op === 'in') return sel.length === 0 || sel.some(s => arr.includes(s));
    if (op === 'notIn') return sel.length === 0 || !sel.some(s => arr.includes(s));
    return true;
  }
  if (fieldDef.type === 'number'){
    const n = Number(value);
    if (op === 'empty')    return value == null;
    if (op === 'notEmpty') return value != null;
    if (v === '' || v == null) return true;
    const a = Number(v);
    if (op === 'eq')  return n === a;
    if (op === 'ne')  return n !== a;
    if (op === 'gt')  return n > a;
    if (op === 'gte') return n >= a;
    if (op === 'lt')  return n < a;
    if (op === 'lte') return n <= a;
    if (op === 'between'){
      if (v2 === '' || v2 == null) return true;
      const b = Number(v2);
      const min = Math.min(a,b), max = Math.max(a,b);
      return n >= min && n <= max;
    }
    return true;
  }
  const s = String(value == null ? '' : value).toLowerCase();
  if (op === 'empty')    return s.trim() === '';
  if (op === 'notEmpty') return s.trim() !== '';
  if (v == null || String(v).trim() === '') return true;
  const needle = String(v).toLowerCase();
  if (op === 'contains')    return s.includes(needle);
  if (op === 'notContains') return !s.includes(needle);
  if (op === 'equals')      return s === needle;
  if (op === 'notEquals')   return s !== needle;
  if (op === 'startsWith')  return s.startsWith(needle);
  if (op === 'endsWith')    return s.endsWith(needle);
  return true;
}

function matchesAdvanced(record, listId){
  const state = advFilters[listId];
  const conds = state.appliedConditions;
  if (!conds || conds.length === 0) return true;
  const results = conds.map(c => matchesCondition(record, c));
  return state.appliedJoin === 'or' ? results.some(Boolean) : results.every(Boolean);
}

function toggleFilterPanel(listId){
  const panel = document.getElementById('af-panel-' + listId);
  if (!panel) return;
  const isOpen = filterPanelOpen[listId];
  if (!isOpen){
    const state = advFilters[listId];
    state.conditions = JSON.parse(JSON.stringify(state.appliedConditions || []));
    state.join = state.appliedJoin || 'and';
    panel.style.display = '';
    filterPanelOpen[listId] = true;
    renderFilterPanel(listId);
  } else {
    panel.style.display = 'none';
    filterPanelOpen[listId] = false;
  }
}

function renderFilterConditions(listId){
  const state = advFilters[listId];
  const fields = AF_FIELDS[listId];
  const wrap = document.getElementById('filter-conditions-' + listId);
  if (!wrap) return;
  wrap.innerHTML = '';
  if (state.conditions.length === 0){
    wrap.innerHTML = '<div class="filter-empty">Условий нет. Нажмите «Добавить условие».</div>';
    return;
  }
  state.conditions.forEach(c => {
    const fieldDef = fields.find(f => f.id === c.field) || fields[0];
    const ops = AF_OPS[fieldDef.type];
    const opDef = ops.find(o => o.id === c.op) || ops[0];

    const row = document.createElement('div');
    row.className = 'filter-row';

    const fieldSel = document.createElement('select');
    fieldSel.className = 'input';
    fields.forEach(f => {
      const opt = document.createElement('option');
      opt.value = f.id;
      opt.textContent = f.label;
      if (f.id === fieldDef.id) opt.selected = true;
      fieldSel.appendChild(opt);
    });
    fieldSel.onchange = () => onFilterFieldChange(listId, c.id, fieldSel.value);

    const opSel = document.createElement('select');
    opSel.className = 'input';
    ops.forEach(o => {
      const opt = document.createElement('option');
      opt.value = o.id;
      opt.textContent = o.label;
      if (o.id === opDef.id) opt.selected = true;
      opSel.appendChild(opt);
    });
    opSel.onchange = () => onFilterOpChange(listId, c.id, opSel.value);

    const valWrap = document.createElement('div');
    valWrap.className = 'filter-value-wrap';
    if (!opDef.noValue){
      if (fieldDef.type === 'enum'){
        const chips = document.createElement('div');
        chips.className = 'filter-enum-chips';
        const selected = Array.isArray(c.value) ? c.value : (c.value ? [c.value] : []);
        fieldDef.options.forEach(opt => {
          const lbl = document.createElement('label');
          const on = selected.includes(opt);
          lbl.className = 'filter-enum-chip' + (on ? ' on' : '');
          lbl.innerHTML = `<input type="checkbox" ${on ? 'checked' : ''} /><span>${esc(opt || '—')}</span>`;
          lbl.onclick = (e) => {
            e.preventDefault();
            const arr = Array.isArray(c.value) ? c.value.slice() : [];
            const i = arr.indexOf(opt);
            if (i >= 0) arr.splice(i,1); else arr.push(opt);
            c.value = arr;
            renderFilterConditions(listId);
          };
          chips.appendChild(lbl);
        });
        valWrap.appendChild(chips);
      } else if (opDef.two){
        const inp1 = document.createElement('input');
        inp1.className = 'input filter-value';
        inp1.type = 'number';
        inp1.placeholder = 'от';
        inp1.value = c.value ?? '';
        inp1.oninput = () => { c.value = inp1.value; };
        const inp2 = document.createElement('input');
        inp2.className = 'input filter-value';
        inp2.type = 'number';
        inp2.placeholder = 'до';
        inp2.value = c.value2 ?? '';
        inp2.oninput = () => { c.value2 = inp2.value; };
        valWrap.appendChild(inp1);
        valWrap.appendChild(inp2);
      } else {
        const inp = document.createElement('input');
        inp.className = 'input filter-value';
        if (fieldDef.type === 'number') inp.type = 'number';
        inp.placeholder = 'значение';
        inp.value = c.value ?? '';
        inp.oninput = () => { c.value = inp.value; };
        valWrap.appendChild(inp);
      }
    } else {
      const note = document.createElement('span');
      note.style.color = 'var(--muted)';
      note.style.fontSize = '11.5px';
      note.textContent = 'значение не требуется';
      valWrap.appendChild(note);
    }

    const del = document.createElement('button');
    del.type = 'button';
    del.className = 'filter-del';
    del.title = 'Удалить условие';
    del.textContent = '✕';
    del.onclick = () => removeFilterCondition(listId, c.id);

    row.appendChild(fieldSel);
    row.appendChild(opSel);
    row.appendChild(valWrap);
    row.appendChild(del);

    wrap.appendChild(row);
  });
}

function addFilterCondition(listId){
  const state = advFilters[listId];
  const fields = AF_FIELDS[listId];
  const firstField = fields[0];
  const firstOp = AF_OPS[firstField.type][0];
  state.conditions.push({
    id: newAfId(),
    field: firstField.id,
    op: firstOp.id,
    value: firstField.type === 'enum' ? [] : '',
    value2: ''
  });
  renderFilterConditions(listId);
}
function removeFilterCondition(listId, condId){
  const state = advFilters[listId];
  state.conditions = state.conditions.filter(c => c.id !== condId);
  renderFilterConditions(listId);
}
function onFilterFieldChange(listId, condId, fieldId){
  const state = advFilters[listId];
  const c = state.conditions.find(x => x.id === condId);
  if (!c) return;
  const fieldDef = AF_FIELDS[listId].find(f => f.id === fieldId);
  const firstOp = AF_OPS[fieldDef.type][0];
  c.field = fieldId;
  c.op = firstOp.id;
  c.value = fieldDef.type === 'enum' ? [] : '';
  c.value2 = '';
  renderFilterConditions(listId);
}
function onFilterOpChange(listId, condId, opId){
  const state = advFilters[listId];
  const c = state.conditions.find(x => x.id === condId);
  if (!c) return;
  c.op = opId;
  renderFilterConditions(listId);
}

function renderAppliedBar(listId){
  const bar = document.getElementById('applied-' + listId);

  if (!conds || conds.length === 0){

  const text = describeConditions(listId, conds, state.appliedJoin);
  bar.style.display = '';

function describeConditions(listId, conds, join){
  const fields = AF_FIELDS[listId];
  return conds.map(c => {
    const fieldDef = fields.find(f => f.id === c.field);
    if (!fieldDef) return '';
    const opDef = AF_OPS[fieldDef.type].find(o => o.id === c.op);
    let val = '';
    if (Array.isArray(c.value)) val = c.value.join(', ');
    else val = c.value ?? '';
    if (opDef && opDef.two && c.value2) val = `${c.value} – ${c.value2}`;
    return `${fieldDef.label} ${opDef ? opDef.label : c.op}${val ? ' ' + val : ''}`;
  }).filter(Boolean).join(join === 'and' ? ' и ' : ' или ');
}

function plural(n, forms){
  const a = Math.abs(n) % 100, b = a % 10;
  if (a > 10 && a < 20) return forms[2];
  if (b > 1 && b < 5) return forms[1];
  if (b === 1) return forms[0];
  return forms[2];
}
function esc(s){
  return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({
    '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
  }[c]));
}

  el.classList.add('on');
  clearTimeout(toast._t);
  toast._t = setTimeout(() => el.classList.remove('on'), 2000);
}

function setupSorting(){
  const attach = (listId, screenId, sortObj, renderFn) => {
    document.querySelectorAll('#' + screenId + ' thead th.sortable').forEach(th => {
      th.onclick = () => {
        const f = th.dataset.sort;
        if (sortObj.field === f){
          if (sortObj.dir === 'asc') sortObj.dir = 'desc';
          else { sortObj.field = null; sortObj.dir = null; }
        } else { sortObj.field = f; sortObj.dir = 'asc'; }
        paintSort('#' + screenId + ' thead', sortObj);
        renderFn();
      };
    });
  };
  attach('fb',  'screen-list',    fbSort,  renderList);
  attach('kb',  'screen-kb-list', kbSort,  renderKbList);
  attach('tpl', 'screen-tpl-list',tplSort, renderTplList);
  attach('ab',  'screen-ab-list', abSort,  renderAbList);
  attach('api', 'screen-api-list',apiSort, renderApiList);
}
function paintSort(sel, sort){
  document.querySelectorAll(sel + ' th.sortable').forEach(th => {
    if (th.dataset.sort === sort.field) th.dataset.dir = sort.dir || '';
    else th.removeAttribute('data-dir');
  });
}
function applySort(rows, sort, accessors){
  if (!sort.field || !sort.dir) return rows;
  const acc = accessors[sort.field];
  if (!acc) return rows;
  const dir = sort.dir === 'asc' ? 1 : -1;
  return rows.slice().sort((a,b) => {
    const va = acc(a), vb = acc(b);
    if (va == null && vb == null) return 0;
    if (va == null) return 1;
    if (vb == null) return -1;
    if (typeof va === 'number' && typeof vb === 'number') return (va - vb) * dir;
    return String(va).localeCompare(String(vb), 'ru') * dir;
  });
}

function selectKbLink(id){
  if (kbSelectedLinkId === id) return;
  kbSelectedLinkId = id;
  document.querySelectorAll('#kb-links-list .link-card').forEach(c => {
    c.classList.toggle('selected', c.dataset.id === id);
  });
  renderKbPreview();
}
function addKbLink(){
  const link = { id:newId('l'), title:'Новая ссылка', url:'https://', description:'', tags:'' };
  kbLinks.push(link);
  kbSelectedLinkId = link.id;
  renderKbLinks();
  toast('Ссылка добавлена');
}
function removeKbLink(id){
  kbLinks = kbLinks.filter(l => l.id !== id);
  if (kbSelectedLinkId === id) kbSelectedLinkId = kbLinks.length ? kbLinks[0].id : null;
  renderKbLinks();
  toast('Ссылка удалена');
}
function updateKbLink(id, key, value){
  const l = kbLinks.find(x => x.id === id);
  if (!l) return;
  l[key] = value;
  if (kbSelectedLinkId !== id) kbSelectedLinkId = id;
  renderKbPreview();
}

function renderKbPreview(){
  const wrap = document.getElementById('kb-preview-link');
  if (!wrap) return;
  const link = kbLinks.find(l => l.id === kbSelectedLinkId);
  if (!link){
    wrap.innerHTML = `<div class="pv-empty-note">Ссылок пока нет — добавьте первую слева, и она появится здесь.</div>`;
    return;
  }
  const tags = String(link.tags || '').split(',').map(s => s.trim()).filter(Boolean);
  wrap.innerHTML = `
    <div class="pv-link">
      <div class="pv-link-title">${esc(link.title || 'Без заголовка')}</div>
      ${link.description ? `<div class="pv-link-desc">${esc(link.description)}</div>` : ''}
      ${link.url ? `<div class="pv-link-url">${esc(link.url.replace(/^https?:\/\//,''))}</div>` : ''}
      ${tags.length ? `<div class="pv-link-tags">${tags.map(t => `<span class="pv-link-tag">${esc(t)}</span>`).join('')}</div>` : ''}
    </div>
  `;
}

function renderTplList(){
  const q = document.getElementById('tpl-q').value.trim().toLowerCase();
  const tpSel = msState['tpl-type'];
  const stSel = msState['tpl-status'];

  const filtered = TPL_DATA.filter(r => {
    if (tpSel.length && !tpSel.includes(r.type)) return false;
    if (stSel.length && !stSel.includes(r.status)) return false;
    if (q.length >= 2){
      const hay = (r.name + ' ' + r.code + ' ' + r.type + ' ' + r.scope + ' ' + (r.text || '')).toLowerCase();
      if (!hay.includes(q)) return false;
    }
    if (!matchesAdvanced(r, 'tpl')) return false;
    return true;
  });
  const rows = applySort(filtered, tplSort, {
    name: r => r.name, type: r => r.type, scope: r => r.scope, status: r => r.status
  });

  const tbody = document.getElementById('tpl-rows');
  const empty = document.getElementById('tpl-empty');
  tbody.innerHTML = '';
  empty.style.display = rows.length === 0 ? 'block' : 'none';

  rows.forEach(r => {
    const tr = document.createElement('tr');
    tr.onclick = () => openTplCard(r);
    const stTag = r.status === 'Активен' ? 'ok'
                : r.status === 'Приостановлен' ? 'warn'
                : r.status === 'Архив' ? 'mute' : 'ink';
    tr.innerHTML = `
      <td class="name-cell"><span class="n">${esc(r.name)}</span><span class="c">${esc(r.code)}</span></td>
      <td>${esc(r.type)}</td>
      <td style="color:var(--muted);font-size:12px">${esc(r.scope)}</td>
      <td><span class="tag ${stTag}">${esc(r.status)}</span></td>
      <td class="right">›</td>
    `;
    tbody.appendChild(tr);
  });

  document.getElementById('tpl-count').textContent =
    rows.length + ' ' + plural(rows.length, ['шаблон','шаблона','шаблонов']);
}

function renderAbSum(){
  const sum = sumAbPercents();
  const sumEl = document.getElementById('ab-sum');
  const valEl = document.getElementById('ab-sum-val');
  const hintEl = document.getElementById('ab-sum-hint');
  if (!sumEl) return;
  valEl.textContent = sum + '%';
  sumEl.classList.toggle('bad', sum !== 100 && abGroups.length > 0);
  sumEl.classList.toggle('ok',  sum === 100);
  if (sum === 100){
    hintEl.textContent = 'Всё сходится';
  } else if (abGroups.length === 0){
    hintEl.textContent = 'Добавьте хотя бы одну группу';
  } else if (sum < 100){
    hintEl.textContent = 'Не хватает ' + (100 - sum) + '%';
  } else {
    hintEl.textContent = 'Перебор на ' + (sum - 100) + '%';
  }
}

function renderAbPreview(){
  const wrap = document.getElementById('ab-preview-dist');
  if (!wrap) return;

  if (abGroups.length === 0){
    wrap.innerHTML = `<div class="pv-empty-note">Пока ни одной группы — добавьте первую слева.</div>`;
    return;
  }

  const normalize = pct => (parseFloat(pct) || 0) / sum * 100;

  const segments = abGroups.map((g, i) => {
    const color = AB_COLORS[i % AB_COLORS.length];
    const width = normalize(g.percent);
    const label = width >= 12 ? Math.round((parseFloat(g.percent) || 0)) + '%' : '';
    return `<div class="ab-dist-seg" style="width:${width}%;background:${color}">${label}</div>`;
  }).join('');

  const legend = abGroups.map((g, i) => {
    const color = AB_COLORS[i % AB_COLORS.length];
    const pct = parseFloat(g.percent) || 0;
    return `
      <div class="ab-dist-row">
        <span class="ab-dot" style="background:${color}"></span>
        <span class="name">${esc(g.name || 'Без названия')}</span>
        <span class="tagcode">${esc(g.tag || '—')}</span>
        <span class="pct">${pct}%</span>
      </div>
    `;
  }).join('');

  wrap.innerHTML = `
    <div class="ab-dist">${segments}</div>
    <div class="ab-dist-legend">${legend}</div>
  `;
}

function methodClass(m){
  const s = String(m || '').toLowerCase();
  if (['get','post','put','patch','delete','head','options'].includes(s)) return s;

function renderApiList(){
  const q  = document.getElementById('api-q').value.trim().toLowerCase();
  const mSel = msState['api-method'];
  const sSel = msState['api-source'];
  const stSel = msState['api-status'];

  const filtered = API_DATA.filter(r => {
    if (apiGroupTab !== 'all' && r.group !== apiGroupTab) return false;
    if (mSel.length && !mSel.includes(r.method)) return false;
    if (sSel.length && !sSel.includes(r.source || '')) return false;
    if (stSel.length && !stSel.includes(r.status)) return false;
    if (q.length >= 2){
      const hay = (r.name + ' ' + r.code + ' ' + r.path + ' ' + r.description).toLowerCase();
      if (!hay.includes(q)) return false;
    }
    if (!matchesAdvanced(r, 'api')) return false;
    return true;
  });
  const rows = applySort(filtered, apiSort, {
    method: r => r.method,
    path: r => r.path,
    group: r => r.group + (r.source ? ' · ' + r.source : ''),
    version: r => r.version,
    status: r => r.status,
    name: r => r.name
  });

  const tbody = document.getElementById('api-rows');
  const empty = document.getElementById('api-empty');
  tbody.innerHTML = '';
  empty.style.display = rows.length === 0 ? 'block' : 'none';

  rows.forEach(r => {
    const tr = document.createElement('tr');
    tr.onclick = () => openApiCard(r);
    const stTag = r.status === 'Активен' ? 'ok'
                : r.status === 'Устарел' ? 'warn'
                : r.status === 'Архив' ? 'mute' : 'ink';
    const groupLine = r.group + (r.source ? ' · ' + r.source : '');
    tr.innerHTML = `

/* ============================================================
   DRAG & DROP (общий)
   ============================================================ */
function attachDrag(card, id, arr, onReorder, stateVar){
  const kind = stateVar || 'field';
  card.addEventListener('dragstart', e => {
    if (kind === 'kbDragId') kbDragId = id;
    else if (kind === 'abDragId') abDragId = id;
    else if (kind === 'apiParamDragId') apiParamDragId = id;
    else dragId = id;
    card.classList.add('dragging');
    e.dataTransfer.effectAllowed = 'move';
    try { e.dataTransfer.setData('text/plain', id); } catch(_){}
  });
  card.addEventListener('dragend', () => {
    card.classList.remove('dragging');
    document.querySelectorAll('.field-card.drop-target, .link-card.drop-target, .ab-group-card.drop-target, .api-edit-row.drop-target')
      .forEach(el => el.classList.remove('drop-target'));
    if (kind === 'kbDragId') kbDragId = null;
    else if (kind === 'abDragId') abDragId = null;
    else if (kind === 'apiParamDragId') apiParamDragId = null;
    else dragId = null;
  });
  card.addEventListener('dragover', e => {
    e.preventDefault();
    e.dataTransfer.dropEffect = 'move';
    const active = kind === 'kbDragId' ? kbDragId
                 : kind === 'abDragId' ? abDragId
                 : kind === 'apiParamDragId' ? apiParamDragId
                 : dragId;
    if (active && active !== id) card.classList.add('drop-target');
  });
  card.addEventListener('dragleave', () => card.classList.remove('drop-target'));
  card.addEventListener('drop', e => {
    e.preventDefault();
    card.classList.remove('drop-target');
    const active = kind === 'kbDragId' ? kbDragId
                 : kind === 'abDragId' ? abDragId
                 : kind === 'apiParamDragId' ? apiParamDragId
                 : dragId;
    if (!active || active === id) return;
    const from = arr.findIndex(x => (x._id || x.id) === active);
    const to   = arr.findIndex(x => (x._id || x.id) === id);
    if (from < 0 || to < 0) return;
    const [moved] = arr.splice(from, 1);
    arr.splice(to, 0, moved);
    if (kind === 'kbDragId') kbDragId = null;
    else if (kind === 'abDragId') abDragId = null;
    else if (kind === 'apiParamDragId') apiParamDragId = null;
    else dragId = null;
    onReorder();
  });
}

let currentExportType = null;

function openExportModal(type){
  const preset = EXPORT_PRESETS[type];
  if (!preset) return;
  currentExportType = type;
  document.getElementById('modal-title').textContent = preset.title;
  document.getElementById('modal-body').innerHTML = `
    <p class="modal-hint">${esc(preset.hint)}</p>
    <div class="modal-fields">
      ${preset.fields.map(f => `
        <label class="modal-check">
          <input type="checkbox" ${f.checked ? 'checked' : ''} data-key="${f.key}" />
          <span>${esc(f.label)}</span>
        </label>
      `).join('')}
    </div>
  `;
  document.getElementById('modal-foot').innerHTML = `
    <button class="btn outline sm" onclick="closeModal()">Отмена</button>
    <button class="btn sm" onclick="doExport()">Выгрузить</button>
  `;
  document.getElementById('modal-backdrop').classList.add('on');
}

function doExport(){
  const selected = [...document.querySelectorAll('#modal-body input[type=checkbox]:checked')]
    .map(cb => cb.dataset.key);
  if (selected.length === 0){
    toast('Выберите хотя бы одно поле');
    return;
  }
  document.getElementById('modal-title').textContent = 'Выгрузка сформирована';
  document.getElementById('modal-body').innerHTML = `
    <p style="margin:0 0 6px;font-size:13px">Файл с выбранными полями готов.</p>
    <p style="margin:0;color:var(--muted);font-size:12px">Проверьте папку загрузок браузера.</p>
  `;
  document.getElementById('modal-foot').innerHTML = `
    <button class="btn sm" onclick="closeModal()">Закрыть</button>
  `;
}

function closeModal(){
  document.getElementById('modal-backdrop').classList.remove('on');
  currentExportType = null;
}
